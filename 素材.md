class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int len = nums.size();
        int left = 0, right = len, mid = (left + right) / 2;
        while(left < right){
            mid = (left + right) / 2;
            if(nums[mid] >= target){
                right = mid;
            }else{
                left = mid + 1;
            }
        }
        if(left >= len || nums[left] != target){
            return vector<int>{-1, -1};
        }
        vector<int> ans;
        ans.push_back(left);
        left = 0, right = len, mid = (left + right) / 2;
        while(left < right){
            mid = (left + right) / 2;
            if(nums[mid] >= target + 1){
                right = mid;
            }else{
                left = mid + 1;
            }
        }
        ans.push_back(left - 1);
        return ans;
    }
};
class Solution {
public:
    int searchInsert(vector<int>& nums, int target) {
        int left = 0, right = nums.size(), mid = (left + right) / 2;
        while(left < right){
            mid = (left + right) / 2;
            if(nums[mid] >= target){
                right = mid;
            }else{
                left = mid + 1;
            }
        }
        return left;
    }
};
class Solution {
public:
    int climbStairs(int n) {
        vector<int> dp(n+1);
        dp[0] = 1;
        dp[1] = 1;
        for(int i = 2; i <= n; i++){
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[n];
    }
};
